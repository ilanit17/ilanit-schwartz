דף עבודה: הָאֲפִיקוֹמָן

קראו את הסיפור בעיון, חשבו היטב וענו על השאלות.

בסיום תוכלו לשמור את הדף עם תשובותיכם ולשלוח למורה.

קריאת הטקסט

הָאֲפִיקוֹמָן

כְּמוֹ בְּכָל שָׁנָה, גַּם הַשָּׁנָה צְפִיתִי לְלֵיל הַסֵּדֶר. אוּלַי הַשָּׁנָה אֶמְצָא סוֹף-סוֹף אֶת הָאֲפִיקוֹמָן!

בְּפֶסַח בַּשָּׁנָה שֶׁעָבְרָה גִיל בֶּן דּוֹד שֶׁלִי מָצָא אֶת הָאֲפִיקוֹמָן וְקִבֵּל אוֹפַנַיִם. בְּפֶסַח שֶׁלְפָנָיו סַבָּא הֶחְבִּיא הַרְבֵּה אֲפִיקוֹמָנִים, כְּדֵי שֶׁכַּלָּם יִמְצְאוּ, אָז זֶה לֹא נֶחְשָׁב.

לִפְנֵי שָׁלוֹשׁ שָׁנִים וִתַּרְתִּי לְאָחִי הַקָּטָן שֶׁבְּכָה הָמוֹן וְעָזַרְתִּי לוֹ לִמְצֹא, וְלִפְנֵי אַרְבַּע שָׁנִים הָיִיתִי מַמָּשׁ קָטָן בְּעַצְמִי.

הַשָּׁנָה תִּכְנַנְתִּי לַעֲקֹב אַחֲרֵי סַבָּא כְּדֵי לִרְאוֹת הֵיכָן הוּא מַחְבִּיא אֶת הָאֲפִיקוֹמָן.

הָאוֹרְחִים הִגִּיעוּ וְלֵיל הַסֵּדֶר הִתְחִיל. לְאַחַר שֶׁקָּרָאנוּ אֶת סִפּוּר יְצִיאַת מִצְרַיִם וְכֻלָּם שָׁתוּ אֶת כּוֹס הַיַּיִן הַשְּׁנִיָּה, הִגִּיעַ סַבָּא לְ"מוֹצִיא מַצָּה", אַחַר כָּךְ בָּצַע אֶת הַמַּצָּה לִשְׁנַיִם, וּפִתְאוֹם אָמַר שֶׁהוּא הוֹלֵךְ לַשֵּׁרוּתִים. "לְבַד!" הוּא הִדְגִּישׁ. הֵבַנְתִּי שֶׁזֶּה הָרֶגַע שֶׁלּוֹ חִכִּיתִי.

עָמַדְתִּי בִּקְצֵה הַמִּסְדְרוֹן וְעָקַבְתִּי אַחֲרָיו בְּמַבָּטִי. רָאִיתִי שֶׁסַבָּא מַכְנִיס מַשֶּׁהוּ לַאֲרוֹן הַנַּעֲלַיִם. זֶה הָיָה הָאֲפִיקוֹמָן!

הָאֲפִיקוֹמָן בְּתוֹךְ נַעֲלֵי הַבַּיִת שֶׁל סָבְתָא!

חִכִּיתִי בְּמֶתַח לְסִיּוּם הָאֲרוּחָה כְּדֵי שֶׁאוּכַל כְּבָר לִמְצֹא אֶת הָאֲפִיקוֹמָן. כְּשֶׁסַבָּא נָתַן אֶת הָאוֹת, כֻּלָּם קָמוּ מֵהַכִּסְאוֹת בִּמְהִירוּת וְהֵחֵלּוּ לְהִתְרוֹצֵץ בְּכָל הַבַּיִת.

מִהַרְתִּי לַאֲרוֹן הַנַּעֲלַיִם, הוֹצֵאתִי אֶת הָאֲפִיקוֹמָן וְקָרָאתִי בְּקוֹל: "מָצָאתִי! מָצָאתִי אֶת הָאֲפִיקוֹמָן"!

כֻּלָּם מָחֲאוּ לִי כַּפַּיִם. אֲנִי הִסְמַקְתִּי, הִשְׁפַּלְתִּי אֶת מַבָּטִי וְחִיַּכְתִּי.

סַבָּא נָשַׁק לִי וְאָמַר בַּחֲגִיגִיּוּת: "כָּל הַכָּבוֹד, עֹמֶר! בְּסִיּוּם הַחַג נֵלֵךְ לִקְנוֹת לְךָ מַתָּנָה".

בְּתֹם חֻפְשַׁת הַפֶּסַח סַבָּא וַאֲנִי הָלַכְנוּ לִקְנוֹת אֶת הַמַּתָּנָה. בַּדֶּרֶךְ הִתְכַּסּוּ עֵינַי בִּדְמָעוֹת, הִתְאַפַּקְתִּי שֶׁלֹּא לִבְכּוֹת. נֶעֱצַרְתִּי בַּכְּנִיסָה לַחֲנוּת, וְהַדְמָעוֹת הֵחֵלּוּ לִזְלֹג.

סַבָּא הִבִּיט בִּי בְּעֵינַיִם שׁוֹאֲלוֹת. סִפַּרְתִּי לְסַבָּא בִּמְהִירוּת אֵיךְ מָצָאתִי אֶת הָאֲפִיקוֹמָן. עָצַרְתִּי אֶת נְשִׁימָתִי וְהִבַּטְתִּי בְּסַבָּא. סַבָּא חִבֵּק אוֹתִי חָזָק וַאֲנִי נָשַׁמְתִּי לִרְוָחָה.

כְּשֶׁפָּנִיתִי לָצֵאת אֶל הָרְחוֹב סַבָּא שָׁאַל: "עֹמֶר, לֹא שָׁכַחְתָּ מַשֶּׁהוּ? מָה עִם הַלֶּגוֹ?"

חִיַּכְתִּי לְסַבָּא וְנִכְנַסְנוּ יָד בְּיָד אֶל הַחֲנוּת.
לֹא יָדַעְתִּי מִמָּה לִשְׂמֹחַ יוֹתֵר: מֵהַלֶּגוֹ הֶחָדָשׁ אוֹ מֵהַהֲקָלָה שֶׁהִרְגַּשְׁתִּי אַחֲרֵי שֶׁסִּפַּרְתִּי לְסַבָּא אֶת הַכֹּל.

1

בַּשּׁוּרָה הָרִאשׁוֹנָה מוֹפִיעַ הַמִּשְׁפָּט: "כְּמוֹ בְּכָל שָׁנָה, גַּם הַשָּׁנָה צְפִיתִי לְלֵיל הַסֵּדֶר". בִּמְקוֹם הַפֹּעַל "צְפִיתִי" נִתָּן לִכְתֹּב:

2

בְּשׁוּרוֹת 3-7 מְתָאֵר עֹמֶר מָה קָרָה בַּשָּׁנִים הַקוֹדְמוֹת. הַתְאִימוּ אֶת הַזְּמַן לָאֵרוּעַ מִתּוֹךְ הָרְשִׁימָה (בחרו את התשובה הנכונה לכל זמן):

לִפְנֵי אַרְבַּע שָׁנִים:
לִפְנֵי שָׁלוֹשׁ שָׁנִים:
בְּפֶסַח שֶׁלְפָנָיו (לפני שנתיים):
בְּפֶסַח בַּשָּׁנָה שֶׁעָבְרָה:
3

מַדּוּעַ, לְדַעְתְּכֶם, עֹמֶר חָשַׁב שֶׁ"אֵין זֶה נֶחְשָׁב" כְּשֶׁסַבָּא הֶחְבִּיא הַרְבֵּה אֲפִיקוֹמָנִים?

4

מָה הָיָה הָרֶמֶז שֶׁל סַבָּא כְּשֶׁהוּא יָצָא לְהַחְבִּיא אֶת הָאֲפִיקוֹמָן? (היעזרו בטקסט)

5

אֵיפֹה הִטְמִין (החביא) סַבָּא אֶת הָאֲפִיקוֹמָן?

6

מָה פֵּרוּשׁ הַמִּלָּה "בָּצַע" בַּמִּשְׁפָּט "אַחַר כָּךְ בָּצַע אֶת הַמַּצָּה לִשְׁנַיִם"?

7

בַּטֶקְסְט נֶאֱמַר: "כְּשֶׁסַבָּא נָתַן אֶת הָאוֹת, כֻּלָּם קָמוּ מֵהַכִּסְאוֹת בִּמְהִירוּת". אֵיזוֹ מִלָּה יְכוֹלָה לְהַחְלִיף אֶת הַמִּלָּה "אוֹת"?

8

מַדּוּעַ בָּכָה עֹמֶר כְּשֶׁהָלַךְ עִם סַבָּא לִקְנוֹת אֶת הַמַּתָּנָה?

9

בְּשׁוּרָה 27 כָּתוּב: "עָצַרְתִּי אֶת נְשִׁימָתִי וְהִבַּטְתִּי בְּסַבָּא". מָה פֵּרוּשׁ הַבִּטּוּי?

10

כָּתוּב: "סַבָּא חִבֵּק אוֹתִי חָזָק וַאֲנִי נָשַׁמְתִּי לִרְוָחָה." מַדּוּעַ נָשַׁם עֹמֶר לִרְוָחָה (נרגע, הוקל לו)?

11

שְׁאֵלַת רֶצֶף - סַדְּרוּ אֶת הָאֵרוּעִים הַבָּאִים לְפִי סֵדֶר הִתְרַחֲשׁוּתָם בַּסִּפּוּר (בחרו מספרים 1 עד 6):

עֹמֶר מוֹצֵא אֶת הָאֲפִיקוֹמָן.
עֹמֶר וְסַבָּא יוֹצְאִים לִקְנוֹת אֶת הַמַּתָּנָה.
עֹמֶר עוֹקֵב אַחֲרֵי סַבָּא וּמְגַלֶּה שֶׁהוּא מַחְבִּיא אֶת הָאֲפִיקוֹמָן בְּנַעֲלֵי הַבַּיִת שֶׁל סָבְתָא.
עֹמֶר מְגַלֶּה לְסַבָּא שֶׁעָקַב אַחֲרָיו וּמִתְרַגֵּשׁ מִתְּגוּבָתוֹ שֶׁל סַבָּא.
סַבָּא מַכְרִיז שֶׁהוּא יוֹצֵא לַשֵּׁרוּתִים וּמַדְגִּישׁ שֶׁהוּא רוֹצֶה לָלֶכֶת לְבַד.
לְאַחַר קְרִיאַת סִפּוּר יְצִיאַת מִצְרַיִם וּשְׁתִיַּת הַכּוֹס הַשְּׁנִיָּה, הִגִּיעַ זְמַן הַסְתָּרַת הָאֲפִיקוֹמָן.
12

הַשְׁלִימוּ אֶת הַקֶּטַע הַבָּא בְּעֶזְרַת הַמִּלִּים בְּמַחְסַן הַמִּלִּים:

מחסן מילים: רָאָה | מִרְמָה | שָׂמַח | לֶגוֹ | הִתְוַדָּה | יִמְצָא | הֶחְבִּיא | הֵצִיק
עֹמֶר תִּכְנַן הֵיטֵב אֶת הָאֹפֶן בּוֹ אֶת הָאֲפִיקוֹמָן הַשָּׁנָה. כַּאֲשֶׁר סַבָּא קָם לָלֶכֶת לַשֵּׁרוּתִים, עֹמֶר עָמַד בַּמִּסְדְרוֹן ו- אֶת סַבָּא פּוֹתֵחַ אֶת אֲרוֹן הַנַּעֲלַיִם, ו- אֶת הָאֲפִיקוֹמָן בְּתוֹךְ נַעֲלֵי הַבַּיִת שֶׁל סָבְתָא. כַּאֲשֶׁר עֹמֶר מָצָא אֶת הָאֲפִיקוֹמָן, הוּא הָיָה מְאוֹד.

כְּכָל הַנִּרְאֶה, מַעֲשֵׂה ה- שֶׁעָשָׂה עֹמֶר לוֹ וְלִפְנֵי שֶׁהוּא וְסַבָּא נִכְנְסוּ לַחֲנוּת לִקְנוֹת אֶת ה- , עֹמֶר בִּפְנֵי סַבָּא שֶׁעָקַב אַחֲרָיו וְכָךְ גִּלָּה הֵיכָן הָאֲפִיקוֹמָן.

סיימתם? כל הכבוד!

עכשיו תוכלו לשמור ולהוריד את דף העבודה עם התשובות שלכם ולשלוח למורה.
