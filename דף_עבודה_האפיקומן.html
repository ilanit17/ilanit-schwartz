דַּף עֲבוֹדָה – "הָאֲפִיקוֹמָן"

הֲסָקַת מַסְקָנוֹת, מִילוֹת קִישׁוּר וּמַמְדֵּי הֲבָנָה

סיפור מאת אילנית שוורץ | כיתה ד

⚠️ יש להזין שם לפני שמירה או שליחה

📖 טֶקְסט הַמָּקוֹר

1 כְּמוֹ בְּכָל שָׁנָה, גַּם הַשָּׁנָה צִפִּיתִי לְלֵיל הַסֵּדֶר. אוּלַי הַשָּׁנָה אֶמְצָא סוֹף-סוֹף

אֶת הָאֲפִיקוֹמָן!

3 אוֹ שֶׁלֹּא. בְּפֶסַח שֶׁל לְפָנָיו גִּיל בֶּן דּוֹד שֶׁלִּי מָצָא אֶת הָאֲפִיקוֹמָן וְקִיבֵּל אוֹפַנַּיִם.

וּבְפֶסַח שֶׁעָבַר, שֶׁכֻּלָּם יִמְצְאוּ, אָז זֶה סָבָא הֶחְבִּיא הַרְבֵּה אֲפִיקוֹמָנִים, כְּדֵי

5 שֶׁכֻּלָּם יִמְצְאוּ, אָז זֶה לֹא נֶחְשָׁב.

וִיתַּרְתִּי לְאָחִי הַקָּטָן שֶׁבָּכָה וְעָזַרְתִּי לוֹ לִמְצֹא, וְלִפְנֵי חָמֵשׁ שָׁנִים

הָיִיתִי מַמָּשׁ קָטָן בְּעַצְמִי.

8 הַשָּׁנָה הִתְכַּנַּנְתִּי לַעֲקֹב אַחֲרֵי סָבָא כְּדֵי לִרְאוֹת הֵיכָן הוּא מַחְבִּיא אֶת הָאֲפִיקוֹמָן.

וְלֵיל הַסֵּדֶר הִגִּיעַ וְהִתְחִיל. לְאַחַר שֶׁקָרָאנוּ אֶת סִיפּוּר יְצִיאַת מִצְרַיִם וְכֻלָּם

10 שָׁתוּ הַיַּיִן, הִגִּיעַ סָבָא לְ"מוֹצִיא מַצָּה", אַחַר כָּךְ בָּצַע אֶת הַמַּצָּה לִשְׁנַיִם,

וּפִתְאוֹם אָמַר שֶׁהוּא הוֹלֵךְ לַשֵּׁירוּתִים. "לְבַד!" הוּא הִדְגִּישׁ. הֵבַנְתִּי שֶׁזֶּה הָרֶגַע

שֶׁלִּי.

13 חִיכִּיתִי רֶגַע. עָמַדְתִּי בִּקְצֵה הַמִּסְדְּרוֹן וְעָקַבְתִּי אַחֲרָיו בַּמַּבָּטִי. רָאִיתִי שֶׁסָּבָא

פּוֹתֵחַ אֶת אֲרוֹן הַנַּעֲלַיִם. זֶה הָיָה הָאֲפִיקוֹמָן!

15 הָאֲפִיקוֹמָן בְּתוֹךְ נַעֲלֵי הַבַּיִת שֶׁל סָבְתָא!

חִיכִּיתִי בְּמֶתַח וּכְשֶׁסָּבָא נָתַן אֶת הָאוֹת, כֻּלָּם קָמוּ מֵהַכִּיסָאוֹת בִּמְהִירוּת

וְהֵחֵלּוּ לְהִתְרוֹצֵץ בְּכָל הַבַּיִת. מִיהַרְתִּי לַאֲרוֹן הַנַּעֲלַיִם, הוֹצֵאתִי אֶת הָאֲפִיקוֹמָן

וְקָרָאתִי בְקוֹל: "מָצָאתִי! מָצָאתִי אֶת הָאֲפִיקוֹמָן!"

20 כֻּלָּם מָחֲאוּ לִי כַּפַּיִם.

אֲנִי הִסְמַקְתִּי, הִשְׁפַּלְתִּי אֶת מַבָּטִי וְחִיַּכְתִּי. סָבָא נָשַׁק לִי וְאָמַר בַּחֲגִיגִיּוּת:

"כָּל הַכָּבוֹד, עֹמֶר! בְּסִיּוּם הַחַג נֵלֵךְ לִקְנוֹת לְךָ מַתָּנָה".

23 בְּתֹם חֻפְשַׁת הַפֶּסַח סָבָא וַאֲנִי הָלַכְנוּ לִקְנוֹת אֶת הַמַּתָּנָה. בַּדֶּרֶךְ הִתְכַּסּוּ עֵינַי

בִּדְמָעוֹת, הִתְאַפַּקְתִּי שֶׁלֹּא לִבְכּוֹת. נֶעֱצַרְתִּי בַּכְּנִיסָה לַחֲנוּת, וְהַדְּמָעוֹת הֵחֵלּוּ לִזְלֹג.

25 סָבָא הִבִּיט בִּי בְּעֵינַיִם שׁוֹאֲלוֹת.

סִיפַּרְתִּי לַסָּבָא בִּמְהִירוּת אֵיךְ מָצָאתִי אֶת הָאֲפִיקוֹמָן.

27 עָצַרְתִּי אֶת נְשִׁימָתִי וְהִבַּטְתִּי בַּסָּבָא. סָבָא חִיבֵּק אוֹתִי חָזָק וַאֲנִי נָשַׁמְתִּי

לִרְוָחָה.

כְּשֶׁפָּנִיתִי לָצֵאת אֶל הָרְחוֹב סָבָא שָׁאַל: "עֹמֶר, לֹא שָׁכַחְתָּ מַשֶּׁהוּ? מָה עִם הַלֶּגוֹ?"

30 חִיַּכְתִּי לַסָּבָא וְנִכְנַסְנוּ יָד בְּיָד אֶל הַחֲנוּת.

לֹא יָדַעְתִּי מִמָּה לִשְׂמֹחַ יוֹתֵר: מֵהַלֶּגוֹ הֶחָדָשׁ אוֹ מֵהַהֲקָלָה שֶׁהִרְגַּשְׁתִּי

אַחֲרֵי שֶׁסִּיפַּרְתִּי לַסָּבָא אֶת הַכֹּל.

הטקסט מתוך: גלים פרו מבית סנונית ועת הדעת

1

בַּשׁוּרָה הָרִאשׁוֹנָה מוֹפִיעַ הַמִּשְׁפָּט: "כְּמוֹ בְּכָל שָׁנָה, גַּם הַשָּׁנָה צִפִּיתִי לְלֵיל הַסֵּדֶר". בִּמְקוֹם הַפֹּעַל "צִפִּיתִי" נִתָּן לִכְתֹּב:

2

בְּשׁוּרוֹת 3–7, מְתָאֵר עֹמֶר מָה קָרָה וּמִי מָצָא אֶת הָאֲפִיקוֹמָן בַּשָּׁנִים הַקּוֹדְמוֹת. הַתְאִימוּ עַל פִּי הַכָּתוּב בֵּין הַזְּמָן לְבֵין הָאֵירוּעַ הַקָּשׁוּר בָּאֲפִיקוֹמָן.

הַזְּמָן ↔ אֵירוּעַ בְּנוֹשֵׂא הָאֲפִיקוֹמָן
לִפְנֵי אַרְבַּע שָׁנִים א. גִּיל בֶּן דּוֹד שֶׁלִּי מָצָא אֶת הָאֲפִיקוֹמָן וְקִיבֵּל אוֹפַנַּיִם.
לִפְנֵי שָׁלוֹשׁ שָׁנִים ב. סָבָא הֶחְבִּיא הַרְבֵּה אֲפִיקוֹמָנִים, כְּדֵי שֶׁכֻּלָּם יִמְצְאוּ, אָז זֶה לֹא נֶחְשָׁב.
בְּפֶסַח שֶׁלְּפָנָיו (לִפְנֵי שְׁנָתַיִם) ג. הָיִיתִי מַמָּשׁ קָטָן בְּעַצְמִי.
בְּפֶסַח בַּשָּׁנָה שֶׁעָבְרָה ד. וִיתַּרְתִּי לְאָחִי הַקָּטָן שֶׁבָּכָה וְעָזַרְתִּי לוֹ לִמְצֹא.
3

מַדּוּעַ, לְדַעְתְּכֶם, עֹמֶר חָשַׁב שֶׁ"אֵין זֶה נֶחְשָׁב" כְּשֶׁסָּבָא הֶחְבִּיא הַרְבֵּה אֲפִיקוֹמָנִים?

4

מָה תִּכְנֵן עֹמֶר לַעֲשׂוֹת הַשָּׁנָה כְּדֵי לִמְצֹא אֶת הָאֲפִיקוֹמָן?

5

מַדּוּעַ רָצָה עֹמֶר כָּל כָּךְ לִמְצֹא אֶת הָאֲפִיקוֹמָן?

6

מָה פֵּרוּשׁ הַמִּלָּה "בָּצַע" בַּמִּשְׁפָּט "אַחַר כָּךְ בָּצַע אֶת הַמַּצָּה לִשְׁנַיִם" (שׁוּרָה 10)?

7

מָה הָיָה הָרֶמֶז שֶׁל סָבָא כְּשֶׁהוּא יָצָא לְהַחְבִּיא אֶת הָאֲפִיקוֹמָן?

8

אֵיפֹה הִטְמִין סָבָא אֶת הָאֲפִיקוֹמָן?

9

בַּטֶּקְסְט נֶאֱמַר: "כְּשֶׁסָּבָא נָתַן אֶת הָאוֹת, כֻּלָּם קָמוּ מֵהַכִּיסָאוֹת בִּמְהִירוּת וְהֵחֵלּוּ לְהִתְרוֹצֵץ בְּכָל הַבַּיִת".

אֵיזוֹ מִלָּה יְכוֹלָה לְהַחֲלִיף אֶת הַמִּלָּה "אוֹת"?

10

מַדּוּעַ בָּכָה עֹמֶר כְּשֶׁהָלַךְ עִם סָבָא לִקְנוֹת אֶת הַמַּתָּנָה?

11

בְּשׁוּרָה 27 כָּתוּב: "עָצַרְתִּי אֶת נְשִׁימָתִי וְהִבַּטְתִּי בַּסָּבָא".

מָה פֵּרוּשׁ הַבִּיטוּי 'עָצַרְתִּי אֶת נְשִׁימָתִי'?

12

בְּשׁוּרָה 27 כָּתוּב: "סָבָא חִיבֵּק אוֹתִי חָזָק וַאֲנִי נָשַׁמְתִּי לִרְוָחָה."

מַדּוּעַ נָשַׁם עֹמֶר לִרְוָחָה (נִרְגַּע, הוּקַל לוֹ) כַּאֲשֶׁר סָבָא חִיבֵּק אוֹתוֹ?

13

שְׁאֵלַת רֶצֶף – סַדְּרוּ אֶת הָאֵירוּעִים הַבָּאִים לְפִי סֵדֶר הִתְרַחֲשׁוּתָם בַּסִּיפּוּר (1 = קרה ראשון):

14

הַשְׁלִימוּ אֶת הַקֶּטַע הַבָּא בְּעֶזְרַת הַמִּילִים בְּמַחְסָן הַמִּילִים:

מַחְסַן מִילִים:

💡 טיפ: לִחֲצוּ עַל מִילָה מֵהַמַּחְסָן, וְאָז לִחֲצוּ עַל הַמָּקוֹם הָרֵיק הַמַּתְאִים.

✍

מְשִׂימַת כְּתִיבָה – מִכְתָּב לְעֹמֶר

כִּתְבוּ מִכְתָּב לְעֹמֶר, הַדְּמוּת הָרָאשִׁית בַּסִּיפּוּר. הַצִיגוּ אֶת עַצְמְכֶם, שַׁתְּפוּ אוֹתוֹ בְּמָה שֶׁלְּמַדְתֶּם עָלָיו, הַבִּיעוּ אֶת דַּעְתְּכֶם עַל הִתְנַהֲגוּתוֹ, וְהַצִיעוּ לוֹ הַמְלָצוֹת.

מִבְנֶה מוּצָע: 1. פְּנִיָּה לְעֹמֶר  →  2. הַצָּגַת עַצְמְכֶם  →  3. מָה לְמַדְתֶּם עָלָיו  →  4. דַּעְתְּכֶם וְנִמּוּקִים  →  5. הַמְלָצוֹת  →  6. סִיּוּם וַחֲתִימָה
